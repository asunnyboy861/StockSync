#### 3.2.1 库存同步引擎（核心Actor）

```swift
import Foundation

actor InventorySyncEngine {
    private var channels: [SalesChannel] = []
    private var syncQueue: [SyncEvent] = []
    private var isProcessing = false
    
    struct SyncEvent {
        let id: UUID
        let source: ChannelType
        let sku: String
        let quantityChange: Int
        let timestamp: Date
        let orderId: String?
    }
    
    enum ChannelType: String, CaseIterable {
        case shopify
        case woocommerce
        case etsy
        case amazon
        case manual
    }
    
    func addChannel(_ channel: SalesChannel) {
        channels.append(channel)
    }
    
    func processSaleEvent(
        source: ChannelType,
        sku: String,
        quantitySold: Int,
        orderId: String?
    ) async throws {
        let event = SyncEvent(
            id: UUID(),
            source: source,
            sku: sku,
            quantityChange: -quantitySold,
            timestamp: Date(),
            orderId: orderId
        )
        syncQueue.append(event)
        await processQueue()
    }
    
    func processRestockEvent(
        source: ChannelType,
        sku: String,
        quantityRestocked: Int
    ) async throws {
        let event = SyncEvent(
            id: UUID(),
            source: source,
            sku: sku,
            quantityChange: quantityRestocked,
            timestamp: Date(),
            orderId: nil
        )
        syncQueue.append(event)
        await processQueue()
    }
    
    private func processQueue() async {
        guard !isProcessing, !syncQueue.isEmpty else { return }
        isProcessing = true
        defer { isProcessing = false }
        
        while let event = syncQueue.first {
            syncQueue.removeFirst()
            
            do {
                try await syncToAllChannels(event: event)
            } catch {
                syncQueue.append(event)
                try? await Task.sleep(for: .seconds(5))
            }
        }
    }
    
    private func syncToAllChannels(event: SyncEvent) async throws {
        await withThrowingTaskGroup(of: Void.self) { group in
            for channel in channels where channel.type != event.source {
                group.addTask {
                    try await channel.updateInventory(
                        sku: event.sku,
                        change: event.quantityChange
                    )
                }
            }
            try await group.waitForAll()
        }
    }
}
```

#### 3.2.2 Shopify API客户端

```swift
import Foundation
import CryptoKit

final class ShopifyAPIClient: SalesChannel {
    let type: InventorySyncEngine.ChannelType = .shopify
    
    private let storeDomain: String
    private let accessToken: String
    private let session: URLSession
    
    init(storeDomain: String, accessToken: String) {
        self.storeDomain = storeDomain
        self.accessToken = accessToken
        self.session = URLSession.shared
    }
    
    func fetchInventoryLevels() async throws -> [InventoryItem] {
        var request = URLRequest(
            url: URL(string: "https://\(storeDomain)/admin/api/2025-04/inventory_levels.json")!
        )
        request.setValue(accessToken, forHTTPHeaderField: "X-Shopify-Access-Token")
        request.httpMethod = "GET"
        
        let (data, _) = try await session.data(for: request)
        let response = try JSONDecoder().decode(
            ShopifyInventoryResponse.self, from: data
        )
        return response.inventoryLevels.map { level in
            InventoryItem(
                sku: level.sku ?? "",
                available: level.available,
                channelId: "\(storeDomain)-\(level.inventoryItemId)"
            )
        }
    }
    
    func updateInventory(sku: String, change: Int) async throws {
        let itemId = try await lookupInventoryItemId(sku: sku)
        
        var request = URLRequest(
            url: URL(string: "https://\(storeDomain)/admin/api/2025-04/inventory_levels/adjust.json")!
        )
        request.setValue(accessToken, forHTTPHeaderField: "X-Shopify-Access-Token")
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpMethod = "POST"
        
        let body: [String: Any] = [
            "inventory_item_id": itemId,
            "available_adjustment": change
        ]
        request.httpBody = try JSONSerialization.data(withJSONObject: body)
        
        let (_, response) = try await session.data(for: request)
        guard let httpResponse = response as? HTTPURLResponse,
              (200...299).contains(httpResponse.statusCode) else {
            throw SyncError.apiError("Shopify inventory update failed")
        }
    }
    
    func verifyWebhook(data: Data, hmacHeader: String) -> Bool {
        let key = SymmetricKey(data: Data(accessToken.utf8))
        let hmac = HMAC<SHA256>.authenticationCode(for: data, using: key)
        let computedHMAC = Data(hmac).map { String(format: "%02x", $0) }.joined()
        return computedHMAC == hmacHeader
    }
    
    private func lookupInventoryItemId(sku: String) async throws -> String {
        var request = URLRequest(
            url: URL(string: "https://\(storeDomain)/admin/api/2025-04/inventory_items.json?sku=\(sku)")!
        )
        request.setValue(accessToken, forHTTPHeaderField: "X-Shopify-Access-Token")
        
        let (data, _) = try await session.data(for: request)
        let response = try JSONDecoder().decode(
            ShopifyInventoryItemsResponse.self, from: data
        )
        guard let item = response.inventoryItems.first else {
            throw SyncError.skuNotFound(sku)
        }
        return String(item.id)
    }
}
```

#### 3.2.3 WooCommerce API客户端

```swift
import Foundation

final class WooCommerceAPIClient: SalesChannel {
    let type: InventorySyncEngine.ChannelType = .woocommerce
    
    private let storeURL: String
    private let consumerKey: String
    private let consumerSecret: String
    private let session: URLSession
    
    init(storeURL: String, consumerKey: String, consumerSecret: String) {
        self.storeURL = storeURL
        self.consumerKey = consumerKey
        self.consumerSecret = consumerSecret
        self.session = URLSession.shared
    }
    
    func fetchInventoryLevels() async throws -> [InventoryItem] {
        var components = URLComponents(string: "\(storeURL)/wp-json/wc/v3/products")!
        components.queryItems = [
            URLQueryItem(name: "per_page", value: "100"),
            URLQueryItem(name: "status", value: "publish")
        ]
        
        var request = URLRequest(url: components.url!)
        let credentials = Data("\(consumerKey):\(consumerSecret)".utf8).base64EncodedString()
        request.setValue("Basic \(credentials)", forHTTPHeaderField: "Authorization")
        
        let (data, _) = try await session.data(for: request)
        let products = try JSONDecoder().decode([WCProduct].self, from: data)
        
        return products.flatMap { product -> [InventoryItem] in
            if product.variations.isEmpty {
                return [InventoryItem(
                    sku: product.sku ?? "",
                    available: product.stockQuantity ?? 0,
                    channelId: "\(storeURL)-\(product.id)"
                )]
            }
            return []
        }
    }
    
    func updateInventory(sku: String, change: Int) async throws {
        let productId = try await lookupProductId(sku: sku)
        let currentStock = try await getCurrentStock(productId: productId)
        let newStock = currentStock + change
        
        var request = URLRequest(
            url: URL(string: "\(storeURL)/wp-json/wc/v3/products/\(productId)")!
        )
        let credentials = Data("\(consumerKey):\(consumerSecret)".utf8).base64EncodedString()
        request.setValue("Basic \(credentials)", forHTTPHeaderField: "Authorization")
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpMethod = "PUT"
        
        let body: [String: Any] = ["stock_quantity": newStock]
        request.httpBody = try JSONSerialization.data(withJSONObject: body)
        
        let (_, response) = try await session.data(for: request)
        guard let httpResponse = response as? HTTPURLResponse,
              (200...299).contains(httpResponse.statusCode) else {
            throw SyncError.apiError("WooCommerce inventory update failed")
        }
    }
    
    private func lookupProductId(sku: String) async throws -> Int {
        var components = URLComponents(string: "\(storeURL)/wp-json/wc/v3/products")!
        components.queryItems = [URLQueryItem(name: "sku", value: sku)]
        
        var request = URLRequest(url: components.url!)
        let credentials = Data("\(consumerKey):\(consumerSecret)".utf8).base64EncodedString()
        request.setValue("Basic \(credentials)", forHTTPHeaderField: "Authorization")
        
        let (data, _) = try await session.data(for: request)
        let products = try JSONDecoder().decode([WCProduct].self, from: data)
        guard let product = products.first else {
            throw SyncError.skuNotFound(sku)
        }
        return product.id
    }
    
    private func getCurrentStock(productId: Int) async throws -> Int {
        var request = URLRequest(
            url: URL(string: "\(storeURL)/wp-json/wc/v3/products/\(productId)")!
        )
        let credentials = Data("\(consumerKey):\(consumerSecret)".utf8).base64EncodedString()
        request.setValue("Basic \(credentials)", forHTTPHeaderField: "Authorization")
        
        let (data, _) = try await session.data(for: request)
        let product = try JSONDecoder().decode(WCProduct.self, from: data)
        return product.stockQuantity ?? 0
    }
}
```

```swift
import Foundation
import SwiftData

@Model
final class Product {
    @Attribute(.unique) var sku: String
    var name: String
    var imageUrl: String?
    var channels: [ChannelStock]
    var lastSyncAt: Date
    var lowStockThreshold: Int
    var createdAt: Date
    
    var totalAvailable: Int {
        channels.reduce(0) { $0 + $1.available }

    var isLowStock: Bool {
```

#### 3.2.5 Supabase Edge Function（Webhook接收器）

```typescript
// supabase/functions/shopify-webhook/index.ts
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'
import { serve } from 'https://deno.land/std@0.168.0/http/server.ts'

const SHOPIFY_SECRET = Deno.env.get('SHOPIFY_WEBHOOK_SECRET')!

serve(async (req) => {
  if (req.method !== 'POST') {
    return new Response('Method not allowed', { status: 405 })
  }

  const body = await req.text()
  const hmac = req.headers.get('X-Shopify-Hmac-Sha256')

  const key = await crypto.subtle.importKey(
    'raw',
    new TextEncoder().encode(SHOPIFY_SECRET),
    { name: 'HMAC', hash: 'SHA-256' },
    false,
    ['sign']
  )
  const signature = await crypto.subtle.sign('HMAC', key, new TextEncoder().encode(body))
  const computedHmac = Array.from(new Uint8Array(signature))
    .map(b => b.toString(16).padStart(2, '0'))
    .join('')

  if (computedHmac !== hmac) {
    return new Response('Unauthorized', { status: 401 })
  }

  const topic = req.headers.get('X-Shopify-Topic')
  const payload = JSON.parse(body)

  const supabase = createClient(
    Deno.env.get('SUPABASE_URL')!,
    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  )

  if (topic === 'orders/create') {
    for (const lineItem of payload.line_items) {
      await supabase.from('sync_events').insert({
        source: 'shopify',
        store_id: payload.order_id,
        sku: lineItem.sku,
        quantity_change: -lineItem.quantity,
        event_type: 'sale',
        raw_payload: payload
      })
    }
  } else if (topic === 'orders/cancelled' || topic === 'orders/refunded') {
    for (const refund of payload.refunds ?? []) {
      for (const lineItem of refund.refund_line_items ?? []) {
        await supabase.from('sync_events').insert({
          source: 'shopify',
          store_id: payload.order_id,
          sku: lineItem.line_item.sku,
          quantity_change: lineItem.quantity,
          event_type: 'restock',
          raw_payload: payload
        })
      }
    }
  } else if (topic === 'inventory_levels/update') {
    await supabase.from('sync_events').insert({
      source: 'shopify',
      store_id: payload.inventory_item_id,
      sku: null,
      quantity_change: payload.available,
      event_type: 'inventory_update',
      raw_payload: payload
    })
  }

  return new Response('OK', { status: 200 })
})
```

### 3.3 编写规则

| 规则 | 描述 |
|------|------|
| **架构模式** | MVVM + Repository Pattern，View只负责展示，ViewModel处理业务逻辑，Repository处理数据 |
| **并发模型** | 使用 Swift Actor 保护共享可变状态，所有API调用使用 async/await |
| **错误处理** | 自定义 SyncError 枚举，每个错误都有用户友好的描述和恢复建议 |
| **离线优先** | SwiftData 本地缓存优先，网络恢复后自动同步 |
| **安全** | 所有Token存储在Keychain，API密钥使用CryptoKit加密，不硬编码任何密钥 |
| **API版本** | Shopify API: 2025-04，WooCommerce API: v3，所有API调用带版本号 |
| **重试策略** | 指数退避重试（1s → 2s → 4s → 8s），最多3次，超过则记录失败日志 |
| **限流** | Shopify: 2请求/秒，WooCommerce: 根据服务器响应头动态调整 |
| **日志** | 所有同步操作记录到SyncLog模型，支持按时间/状态/渠道筛选 |
| **测试** | 每个API客户端都有Mock版本用于单元测试，使用Protocol抽象 |

### 8.5 SwiftUI 实现示例（Dashboard卡片）

```swift
import SwiftUI
import SwiftData

struct DashboardView: View {
    @Query private var products: [Product]
    @Query(filter: #Predicate<SyncLog> { $0.status == "failed" })
    private var failedSyncs: [SyncLog]
    
    var totalStock: Int { products.reduce(0) { $0 + $1.totalAvailable } }
    var lowStockItems: [Product] { products.filter(\.isLowStock) }
    
    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(spacing: 16) {
                    statsRow
                    lowStockAlert
                    recentActivity
                }
                .padding()
            }
            .navigationTitle("StockSync")
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button { } label: {
                        Image(systemName: "bell.badge")
                    }
                }
            }
            .refreshable {
                await refreshData()
            }
        }
    }
    
    private var statsRow: some View {
        HStack(spacing: 12) {
            StatCard(
                title: "Total Stock",
                value: "\(totalStock)",
                icon: "cube.box",
                color: .blue
            )
            StatCard(
                title: "Low Stock",
                value: "\(lowStockItems.count)",
                icon: "exclamationmark.triangle",
                color: lowStockItems.isEmpty ? .green : .orange
            )
            StatCard(
                title: "Sync Status",
                value: failedSyncs.isEmpty ? "OK" : "\(failedSyncs.count)",
                icon: failedSyncs.isEmpty ? "checkmark.circle" : "xmark.circle",
                color: failedSyncs.isEmpty ? .green : .red
            )
        }
    }
    
    private var lowStockAlert: some View {
        Group {
            if !lowStockItems.isEmpty {
                VStack(alignment: .leading, spacing: 8) {
                    Label("Low Stock Alert", systemImage: "exclamationmark.triangle.fill")
                        .font(.headline)
                        .foregroundStyle(.orange)
                    
                    ForEach(lowStockItems.prefix(3)) { product in
                        HStack {
                            Circle()
                                .fill(product.totalAvailable == 0 ? Color.red : Color.orange)
                                .frame(width: 8, height: 8)
                            Text(product.name)
                                .font(.subheadline)
                            Spacer()
                            Text("\(product.totalAvailable) left")
                                .font(.subheadline.bold())
                                .foregroundStyle(product.totalAvailable == 0 ? .red : .orange)
                        }
                    }
                }
                .padding()
                .background(.ultraThinMaterial, in: RoundedRectangle(cornerRadius: 12))
            }
        }
    }
    
    private var recentActivity: some View {
        VStack(alignment: .leading, spacing: 8) {
            Text("Recent Sync Activity")
                .font(.headline)
            
            RecentSyncListView()
        }
    }
    
    private func refreshData() async {
        // Trigger full sync from all channels
    }
}

struct StatCard: View {
    let title: String
    let value: String
    let icon: String
    let color: Color
    
    var body: some View {
        VStack(spacing: 6) {
            Image(systemName: icon)
                .font(.title2)
                .foregroundStyle(color)
            Text(value)
                .font(.title.bold())
            Text(title)
                .font(.caption)
                .foregroundStyle(.secondary)
        }
        .frame(maxWidth: .infinity)
        .padding()
        .background(.ultraThinMaterial, in: RoundedRectangle(cornerRadius: 12))
    }
}
```
